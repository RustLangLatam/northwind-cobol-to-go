// db2realrun700 (this two-service build): run the attested NWC700 service of the two-service
// deliverable against real Db2 for i on PUB400 and compare its reports with the golden
// OPM COBOL/400 spooled files.
//
// It creates CUSMST in PRISMIA1 from the upstream DDS (CRTSRCPF + CRTPF via
// QSYS2.QCMDEXC, as the golden run did). For each data set it empties CUSMST, loads the
// golden CSV with one SQL INSERT per row in CSV order, and prints the list through
// application.NewService with the REAL Db2 repository (repositories.NewDb2CustomerRepository
// over repositories.CustomerTable(DB2_LIBRARY) and a svcruntime.Holder of db2access.Open
// with repositories.IsPermanentOpenError: the wiring of nwc700.Build and
// core/cmd/nwc700-services without the gRPC server), the production output.SpoolPrinter
// and a fixed clock. Then it drops everything and lists PRISMIA1 again.
//
// Usage: db2realrun700 run <ddsdir> <golden dir> <spool dir> | cleanup
// The password is read from the first line of <password-file> with os.ReadFile
// and only set in this process's environment for db2access.ConfigFromEnv. Every output
// line goes through out(), which scrubs the password as a last guard.
package main

	"milton_prism/core/services/nwc700/application"

)

const lib = "PRISMIA1"

var (
	pw          string
	queryFailed bool
	tables      = []string{"CUSMST"}
	sets        = []string{"edge", "empty", "pages"}
)

func out(format string, a ...any) {
	s := fmt.Sprintf(format, a...)
	if pw != "" {
		s = strings.ReplaceAll(s, pw, "[redacted]")
	}
	fmt.Println(s)
}

func q(s string) string { return "'" + strings.ReplaceAll(s, "'", "''") + "'" }

// setEnv reads the password and sets the DB2_* variables in this process only.
func setEnv(ccsid string) {
	home, _ := os.UserHomeDir()
	b, err := os.ReadFile(filepath.Join(home, "<password-file>"))
	if err != nil {
		fmt.Println("cannot read the credential file")
		os.Exit(2)
	}
	pw = strings.TrimSpace(strings.SplitN(string(b), "\n", 2)[0])
	if pw == "" {
		fmt.Println("credential file has no first line")
		os.Exit(2)
	}
	for k, v := range map[string]string{db2access.EnvHost: "pub400.com", db2access.EnvPort: "9471",
		db2access.EnvLibrary: lib, db2access.EnvUser: "PRISMIA", db2access.EnvPassword: pw,
		db2access.EnvCommitMode: "*NONE", db2access.EnvCCSID: ccsid} {
		_ = os.Setenv(k, v)
	}
}

func cl(ctx context.Context, db *sql.DB, cmd string) error {
	out("CL> %s", cmd)
	_, err := db.ExecContext(ctx, "CALL QSYS2.QCMDEXC("+q(cmd)+")")
	if err != nil {
		out("  FAILED: %v", err)
	}
	return err
}

func exec(ctx context.Context, db *sql.DB, s string) error {
	_, err := db.ExecContext(ctx, s)
	if err != nil {
		out("  SQL FAILED %q: %v", s, err)
	}
	return err
}

func rows(ctx context.Context, db *sql.DB, s string) ([][]string, error) {
	rs, err := db.QueryContext(ctx, s)
	if err != nil {
		return nil, err
	}
	defer rs.Close()
	cols, _ := rs.Columns()
	var res [][]string
	for rs.Next() {
		v := make([]sql.NullString, len(cols))
		p := make([]any, len(cols))
		for i := range v {
			p[i] = &v[i]
		}
		if err := rs.Scan(p...); err != nil {
			return res, err
		}
		r := make([]string, len(cols))
		for i, x := range v {
			r[i] = "<NULL>"
			if x.Valid {
				r[i] = strings.TrimRight(x.String, " ")
			}
		}
		res = append(res, r)
	}
	return res, rs.Err()
}

func dump(ctx context.Context, db *sql.DB, s string) []string {
	rs, err := rows(ctx, db, s)
	if err != nil {
		queryFailed = true // a failed listing must not pass the cleanup verification
		out("  QUERY FAILED: %v", err)
	}
	var lines []string
	for _, r := range rs {
		l := strings.Join(r, " | ")
		lines = append(lines, l)
		out("  %s", l)
	}
	return lines
}

const objQ = "SELECT OBJNAME, OBJTYPE, OBJATTRIBUTE FROM TABLE(QSYS2.OBJECT_STATISTICS('" + lib + "', '*ALL')) ORDER BY OBJNAME"
const mbrQ = "SELECT SYSTEM_TABLE_NAME, SYSTEM_TABLE_MEMBER FROM QSYS2.SYSPARTITIONSTAT WHERE SYSTEM_TABLE_SCHEMA = '" + lib + "' ORDER BY 1, 2"
const splQ = "SELECT SPOOLED_FILE_NAME, JOB_NAME, FILE_NUMBER FROM QSYS2.OUTPUT_QUEUE_ENTRIES_BASIC WHERE USER_NAME = 'PRISMIA' ORDER BY CREATE_TIMESTAMP"

type state struct{ objs, mbrs, spls []string }

func snapshot(ctx context.Context, db *sql.DB, label string) state {
	out("== %s: objects in %s", label, lib)
	o := dump(ctx, db, objQ)
	out("== %s: members in %s", label, lib)
	m := dump(ctx, db, mbrQ)
	out("== %s: spooled files of PRISMIA", label)
	s := dump(ctx, db, splQ)
	return state{o, m, s}
}

func adminDB(cfg db2access.Config) *sql.DB {
	db, err := ibmi.Driver{}.Open(cfg)
	if err != nil {
		out("FATAL open: %v", err)
		os.Exit(1)
	}
	db.SetMaxOpenConns(1)
	return db
}

func member(ctx context.Context, db *sql.DB, mbr, path string) error {
	data, err := os.ReadFile(path)
	if err != nil {
		return err
	}
	if err := cl(ctx, db, "ADDPFM FILE("+lib+"/ZNWDDS) MBR("+mbr+") SRCTYPE(PF)"); err != nil {
		return err
	}
	alias := lib + ".ZZALIAS"
	if err := exec(ctx, db, "CREATE ALIAS "+alias+" FOR "+lib+".ZNWDDS("+mbr+")"); err != nil {
		return err
	}
	defer exec(ctx, db, "DROP ALIAS "+alias)
	for i, l := range strings.Split(strings.TrimRight(string(data), "\n"), "\n") {
		if len(l) > 80 {
			return fmt.Errorf("%s line %d longer than 80", path, i+1)
		}
		if err := exec(ctx, db, fmt.Sprintf("INSERT INTO %s (SRCSEQ, SRCDAT, SRCDTA) VALUES (%d, 0, %s)", alias, i+1, q(l))); err != nil {
			return err
		}
	}
	return nil
}

func setup(ctx context.Context, db *sql.DB, ddsdir string) error {
	if err := cl(ctx, db, "CRTSRCPF FILE("+lib+"/ZNWDDS) RCDLEN(92)"); err != nil {
		return err
	}
	if err := member(ctx, db, "CUSMST", filepath.Join(ddsdir, "cusmst.pf")); err != nil {
		return err
	}
	if err := cl(ctx, db, "CRTPF FILE("+lib+"/CUSMST) SRCFILE("+lib+"/ZNWDDS) SRCMBR(CUSMST)"); err != nil {
		return err
	}
	dump(ctx, db, "SELECT TABLE_NAME, COLUMN_NAME, DATA_TYPE, LENGTH, NUMERIC_SCALE, CCSID FROM QSYS2.SYSCOLUMNS WHERE TABLE_SCHEMA = '"+lib+"' AND TABLE_NAME = 'CUSMST' ORDER BY ORDINAL_POSITION")
	return nil
}

// load empties CUSMST and inserts the data set's rows in CSV order (arrival sequence).
// Every CSV column is CHAR; the others take their DDS default.
func load(ctx context.Context, db *sql.DB, path string) error {
	if err := exec(ctx, db, "DELETE FROM "+lib+".CUSMST"); err != nil {
		return err
	}
	f, err := os.Open(path)
	if err != nil {
		return err
	}
	recs, err := csv.NewReader(f).ReadAll()
	f.Close()
	if err != nil {
		return err
	}
	for _, r := range recs[1:] {
		vals := make([]string, len(r))
		for i, v := range r {
			vals[i] = q(v)
		}
		if err := exec(ctx, db, fmt.Sprintf("INSERT INTO %s.CUSMST (%s) VALUES (%s)", lib,
			strings.Join(recs[0], ","), strings.Join(vals, ","))); err != nil {
			return err
		}
	}
	dump(ctx, db, "SELECT 'CUSMST', COUNT(*) FROM "+lib+".CUSMST")
	return nil
}

func cleanup(ctx context.Context, db *sql.DB, before state) state {
	out("== cleanup")
	_ = exec(ctx, db, "DROP ALIAS "+lib+".ZZALIAS")
	for _, t := range append(append([]string{}, tables...), "ZNWDDS") {
		_ = cl(ctx, db, "DLTF FILE("+lib+"/"+t+")")
	}
	old := map[string]bool{}
	for _, s := range before.spls {
		old[s] = true
	}
	rs, _ := rows(ctx, db, splQ)
	for _, r := range rs {
		if !old[strings.Join(r, " | ")] {
			_ = cl(ctx, db, "DLTSPLF FILE("+r[0]+") JOB("+r[1]+") SPLNBR("+r[2]+")")
		}
	}
	return snapshot(ctx, db, "after")
}

func same(a, b []string) bool { return strings.Join(a, "\n") == strings.Join(b, "\n") }

// waitStore blocks until the holder has connected or failed permanently.
func waitStore(h *svcruntime.Holder[*db2access.Store]) error {
	deadline := time.Now().Add(2 * time.Minute)
	for time.Now().Before(deadline) {
		if err := h.Err(); err != nil {
			return err
		}
		err := h.Do(func(*db2access.Store) error { return nil })
		if !errors.Is(err, svcruntime.ErrNotReady) {
			return err
		}
		time.Sleep(200 * time.Millisecond)
	}
	return errors.New("store did not connect in 2m")
}

func printAndCompare(ctx context.Context, db *sql.DB, gdir, spoolRoot string) {
	dbCfg, err := db2access.ConfigFromEnv(os.Getenv)
	if err != nil {
		out("FATAL config: %v", err)
		return
	}
	table, err := repositories.CustomerTable(dbCfg.Library)
	if err != nil {
		out("FATAL table: %v", err)
		return
	}
	holder := &svcruntime.Holder[*db2access.Store]{}
	holder.Start(ctx, func(ctx context.Context) (*db2access.Store, error) {
		return db2access.Open(ctx, ibmi.Driver{}, dbCfg)
	}, 5*time.Second, svcruntime.WithPermanentErrors(repositories.IsPermanentOpenError))
	defer holder.Close(context.Background())
	if err := waitStore(holder); err != nil {
		out("FATAL store: %v", err)
		return
	}
	out("store connected (db2access.Open, CCSID %d, commit %s)", dbCfg.CCSID, dbCfg.CommitMode)
	repo := repositories.NewDb2CustomerRepository(holder, table)
	loc, _ := time.LoadLocation("America/Chicago")
	clock := svcruntime.NewClock(loc, func() time.Time { return time.Date(2026, 11, 25, 12, 0, 0, 0, loc) })
	tot := totals{}
	for _, set := range sets {
		if err := load(ctx, db, filepath.Join(gdir, "input-cusmst-"+set+".csv")); err != nil {
			out("set %s: load failed: %v", set, err)
			tot.diff++
			continue
		}
		sdir := filepath.Join(spoolRoot, set)
		_ = os.RemoveAll(sdir)
		_ = os.MkdirAll(sdir, 0o755)
		svc := application.NewService(repo, output.NewSpoolPrinter(sdir), clock)
		res, err := svc.PrintCustomerList(ctx)
		if err != nil {
			out("set %s: Go error %v", set, err)
			tot.diff++
			continue
		}
		compareSet(gdir, set, res, sdir, &tot, out)
	}
	out("TOTAL records compared: %d, identical: %d, differing items: %d", tot.cmp, tot.same, tot.diff)
}

func main() {
	if len(os.Args) < 2 {
		fmt.Println("usage: run <ddsdir> <golden dir> <spool dir> | cleanup")
		os.Exit(2)
	}
	setEnv("273")
	cfg, err := db2access.ConfigFromEnv(os.Getenv)
	if err != nil {
		out("FATAL config: %v", err)
		os.Exit(1)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Minute)
	defer cancel()
	db := adminDB(cfg)
	defer db.Close()
	dump(ctx, db, "SELECT CURRENT USER, CURRENT SCHEMA, CURRENT TIMESTAMP, CURRENT SERVER FROM SYSIBM.SYSDUMMY1")
	dump(ctx, db, "SELECT CCSID, DEFAULT_CCSID FROM TABLE(QSYS2.ACTIVE_JOB_INFO(JOB_NAME_FILTER => '*', DETAILED_INFO => 'ALL')) X")
	switch os.Args[1] {
	case "run":
		before := snapshot(ctx, db, "before")
		after := func() state {
			if err := setup(ctx, db, os.Args[2]); err != nil {
				out("setup failed: %v", err)
				return cleanup(ctx, db, before)
			}
			printAndCompare(ctx, db, os.Args[3], os.Args[4])
			return cleanup(ctx, db, before)
		}()
		out("== cleanup verification: objects %v, members %v, spooled files %v (same as before), every listing query answered %v",
			same(before.objs, after.objs), same(before.mbrs, after.mbrs), same(before.spls, after.spls), !queryFailed)
	case "cleanup":
		cleanup(ctx, db, snapshot(ctx, db, "before"))
	}
}

// --- comparison (NWC700): the goldencompare rules with the NWC700 golden README's strips ---

const footer = "-=* https://pub400.com *=-"

type totals struct{ cmp, same, diff int }

type rec struct {
	page, line int
	ctl        byte
	text       string
}

func (r rec) String() string { return fmt.Sprintf("p%d l%02d |%c%s|", r.page, r.line, r.ctl, r.text) }

// fileRecords parses the file output.Render writes: the first record starts the document
// (page 1 line 1, AFTER ADVANCING PAGE: control '1'), "\n\f" is a new page, n line ends
// advance n lines, and the closing line end is not a record. Render strips trailing
// blanks, so each record is padded back to the 132 bytes of the print record; a longer
// record is kept as it is and so differs.
func fileRecords(s string) (recs []rec) {
	page, line, ctl, i := 1, 1, byte('1'), 0
	for i < len(s) {
		end := strings.IndexByte(s[i:], '\n')
		if end < 0 {
			end = len(s) - i
		}
		tx := s[i : i+end]
		if len(tx) < 132 {
			tx += strings.Repeat(" ", 132-len(tx))
		}
		recs = append(recs, rec{page, line, ctl, tx})
		i += end
		if i >= len(s) {
			return
		}
		if i+1 < len(s) && s[i+1] == '\f' {
			page, line, ctl = page+1, 1, '1'
			i += 2
			continue
		}
		n := 0
		for i < len(s) && s[i] == '\n' {
			n++
			i++
		}
		if i >= len(s) { // the closing line end
			return
		}
		line += n
		ctl = map[int]byte{1: ' ', 2: '0', 3: '-'}[n]
		if ctl == 0 {
			ctl = '?'
		}
	}
	return
}

func goldenRecords(lines []string) (recs []rec) {
	page, line := 0, 1
	for _, l := range lines {
		c, tx := l[0], l[1:]
		switch c {
		case '1':
			page, line = page+1, 1
		case ' ':
			line++
		case '0':
			line += 2
		case '-':
			line += 3
		}
		if strings.TrimSpace(tx) == footer || (c != '1' && strings.TrimSpace(tx) == "") {
			continue
		}
		recs = append(recs, rec{page, line, c, tx})
	}
	return
}

func compareSet(gdir, set string, res application.Result, dir string, tot *totals, say func(string, ...any)) {
	files, _ := os.ReadDir(dir)
	if len(files) != 1 {
		say("set %s: expected 1 report file, got %d", set, len(files))
		tot.diff++
		return
	}
	b, _ := os.ReadFile(filepath.Join(dir, files[0].Name()))
	raw, err := os.ReadFile(filepath.Join(gdir, "nwc700-"+set+".fcfc.txt"))
	if err != nil {
		say("set %s: no golden: %v", set, err)
		tot.diff++
		return
	}
	grecs := fileRecords(string(b))
	crecs := goldenRecords(strings.Split(strings.TrimRight(string(raw), "\n"), "\n"))
	// The golden README: the empty page 1 is IBM i behaviour, not NWC700 output. Strip it
	// and number the pages from the first page NWC700 wrote.
	cEmpty := len(crecs) > 0 && crecs[0].ctl == '1' && strings.TrimSpace(crecs[0].text) == ""
	if cEmpty {
		crecs = crecs[1:]
		for i := range crecs {
			crecs[i].page--
		}
	}
	cPages := 0
	if len(crecs) > 0 {
		cPages = crecs[len(crecs)-1].page
	}
	say("set %s: Go customers=%d pages=%d file %s; golden content pages=%d (empty leading page on the host: %v, stripped)",
		set, res.Customers, res.Pages, files[0].Name()[:strings.Index(files[0].Name(), "-2026")], cPages, cEmpty)
	diffs, cnt := 0, 0
	for i := 0; i < len(crecs) || i < len(grecs); i++ {
		var c, g string
		if i < len(crecs) {
			c = crecs[i].String()
		}
		if i < len(grecs) {
			g = grecs[i].String()
		}
		cnt++
		if c != g {
			diffs++
			say("  set %s rec %d\n    COBOL %s\n    Go    %s", set, i+1, c, g)
		} else {
			tot.same++
		}
	}
	if res.Pages != cPages {
		diffs++
		say("  set %s: page count COBOL %d, Go %d", set, cPages, res.Pages)
	}
	say("set %s: %d COBOL records, %d Go records, %d differing", set, len(crecs), len(grecs), diffs)
	tot.cmp += cnt
	tot.diff += diffs
}
