// goldencompare700 (this two-service build): feeds the three PUB400 NWC700 data sets to the attested
// NWC700 service through an in-memory ports.CustomerRepository, writes the report with the
// production output.SpoolPrinter and compares every record (FCFC control plus 132 bytes),
// with its page and line, against the golden spooled files of the OPM COBOL/400 run.
//
// The in-memory repository honours the ListByCountry contract the way the Db2 repository
// does: CHAR values blank padded to the column width, ORDER BY CMCNTY, CMCITY, CMNAME,
// CMCUST in the EBCDIC (CCSID 37) collating sequence. Every character of the data sets is
// invariant between CCSID 37 and 273.
// Usage: goldencompare700 <golden dir>
package main

import (
	"bytes"
	"context"
	"encoding/csv"
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"strings"
	"time"

	"golang.org/x/text/encoding/charmap"

	"milton_prism/core/services/nwc700/application"
	"milton_prism/core/services/nwc700/domain"
	"milton_prism/core/services/nwc700/infrastructure/output"
	"milton_prism/core/shared/svcruntime"
)

var sets = []string{"edge", "empty", "pages"}

type repo struct{ rows []domain.Customer }

func fit(s string, n int) string { return (s + strings.Repeat(" ", n))[:n] }

func ebcdic(s string) []byte {
	b, err := charmap.CodePage037.NewEncoder().Bytes([]byte(s))
	if err != nil {
		panic(err)
	}
	return b
}

func load(path string) *repo {
	f, err := os.Open(path)
	if err != nil {
		panic(err)
	}
	defer f.Close()
	recs, err := csv.NewReader(f).ReadAll()
	if err != nil {
		panic(err)
	}
	col := map[string]int{}
	for i, h := range recs[0] {
		col[h] = i
	}
	r := &repo{}
	for _, x := range recs[1:] {
		r.rows = append(r.rows, domain.Customer{Number: fit(x[col["CMCUST"]], 5), Name: fit(x[col["CMNAME"]], 40),
			Contact: fit(x[col["CMCONT"]], 30), City: fit(x[col["CMCITY"]], 15), Country: fit(x[col["CMCNTY"]], 15),
			Phone: fit(x[col["CMPHON"]], 24)})
	}
	return r
}

func (r *repo) ListByCountry(_ context.Context, limit int) ([]domain.Customer, error) {
	out := append([]domain.Customer(nil), r.rows...)
	sort.SliceStable(out, func(i, j int) bool {
		a, b := out[i], out[j]
		for _, p := range [][2]string{{a.Country, b.Country}, {a.City, b.City}, {a.Name, b.Name}, {a.Number, b.Number}} {
			if c := bytes.Compare(ebcdic(p[0]), ebcdic(p[1])); c != 0 {
				return c < 0
			}
		}
		return false
	})
	if len(out) > limit {
		out = out[:limit]
	}
	return out, nil
}

func main() {
	gdir := os.Args[1]
	loc, _ := time.LoadLocation("America/Chicago")
	clock := svcruntime.NewClock(loc, func() time.Time { return time.Date(2026, 11, 25, 12, 0, 0, 0, loc) })
	tot := totals{}
	for _, set := range sets {
		sp, _ := os.MkdirTemp("", "spool700")
		svc := application.NewService(load(filepath.Join(gdir, "input-cusmst-"+set+".csv")), output.NewSpoolPrinter(sp), clock)
		res, err := svc.PrintCustomerList(context.Background())
		if err != nil {
			fmt.Printf("set %s: Go error %v\n", set, err)
			tot.diff++
			continue
		}
		compareSet(gdir, set, res, sp, &tot, func(f string, a ...any) { fmt.Printf(f+"\n", a...) })
		_ = os.RemoveAll(sp)
	}
	fmt.Printf("TOTAL records compared: %d, identical: %d, differing items: %d\n", tot.cmp, tot.same, tot.diff)
}

// --- comparison (NWC700): the goldencompare rules with the NWC700 golden README's strips ---

const footer = "-=* https://pub400.com *=-"

type totals struct{ cmp, same, diff int }

type rec struct {
	page, line int
	ctl        byte
	text       string
}

func (r rec) String() string { return fmt.Sprintf("p%d l%02d |%c%s|", r.page, r.line, r.ctl, r.text) }

// fileRecords parses the file output.Render writes: the first record starts the document
// (page 1 line 1, AFTER ADVANCING PAGE: control '1'), "\n\f" is a new page, n line ends
// advance n lines, and the closing line end is not a record. Render strips trailing
// blanks, so each record is padded back to the 132 bytes of the print record; a longer
// record is kept as it is and so differs.
func fileRecords(s string) (recs []rec) {
	page, line, ctl, i := 1, 1, byte('1'), 0
	for i < len(s) {
		end := strings.IndexByte(s[i:], '\n')
		if end < 0 {
			end = len(s) - i
		}
		tx := s[i : i+end]
		if len(tx) < 132 {
			tx += strings.Repeat(" ", 132-len(tx))
		}
		recs = append(recs, rec{page, line, ctl, tx})
		i += end
		if i >= len(s) {
			return
		}
		if i+1 < len(s) && s[i+1] == '\f' {
			page, line, ctl = page+1, 1, '1'
			i += 2
			continue
		}
		n := 0
		for i < len(s) && s[i] == '\n' {
			n++
			i++
		}
		if i >= len(s) { // the closing line end
			return
		}
		line += n
		ctl = map[int]byte{1: ' ', 2: '0', 3: '-'}[n]
		if ctl == 0 {
			ctl = '?'
		}
	}
	return
}

func goldenRecords(lines []string) (recs []rec) {
	page, line := 0, 1
	for _, l := range lines {
		c, tx := l[0], l[1:]
		switch c {
		case '1':
			page, line = page+1, 1
		case ' ':
			line++
		case '0':
			line += 2
		case '-':
			line += 3
		}
		if strings.TrimSpace(tx) == footer || (c != '1' && strings.TrimSpace(tx) == "") {
			continue
		}
		recs = append(recs, rec{page, line, c, tx})
	}
	return
}

func compareSet(gdir, set string, res application.Result, dir string, tot *totals, say func(string, ...any)) {
	files, _ := os.ReadDir(dir)
	if len(files) != 1 {
		say("set %s: expected 1 report file, got %d", set, len(files))
		tot.diff++
		return
	}
	b, _ := os.ReadFile(filepath.Join(dir, files[0].Name()))
	raw, err := os.ReadFile(filepath.Join(gdir, "nwc700-"+set+".fcfc.txt"))
	if err != nil {
		say("set %s: no golden: %v", set, err)
		tot.diff++
		return
	}
	grecs := fileRecords(string(b))
	crecs := goldenRecords(strings.Split(strings.TrimRight(string(raw), "\n"), "\n"))
	// The golden README: the empty page 1 is IBM i behaviour, not NWC700 output. Strip it
	// and number the pages from the first page NWC700 wrote.
	cEmpty := len(crecs) > 0 && crecs[0].ctl == '1' && strings.TrimSpace(crecs[0].text) == ""
	if cEmpty {
		crecs = crecs[1:]
		for i := range crecs {
			crecs[i].page--
		}
	}
	cPages := 0
	if len(crecs) > 0 {
		cPages = crecs[len(crecs)-1].page
	}
	say("set %s: Go customers=%d pages=%d file %s; golden content pages=%d (empty leading page on the host: %v, stripped)",
		set, res.Customers, res.Pages, files[0].Name()[:strings.Index(files[0].Name(), "-2026")], cPages, cEmpty)
	diffs, cnt := 0, 0
	for i := 0; i < len(crecs) || i < len(grecs); i++ {
		var c, g string
		if i < len(crecs) {
			c = crecs[i].String()
		}
		if i < len(grecs) {
			g = grecs[i].String()
		}
		cnt++
		if c != g {
			diffs++
			say("  set %s rec %d\n    COBOL %s\n    Go    %s", set, i+1, c, g)
		} else {
			tot.same++
		}
	}
	if res.Pages != cPages {
		diffs++
		say("  set %s: page count COBOL %d, Go %d", set, cPages, res.Pages)
	}
	say("set %s: %d COBOL records, %d Go records, %d differing", set, len(crecs), len(grecs), diffs)
	tot.cmp += cnt
	tot.diff += diffs
}
