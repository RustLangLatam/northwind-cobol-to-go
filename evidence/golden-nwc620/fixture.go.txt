func cents(c int) string {
	sign := ""
	if c < 0 {
		sign, c = "-", -c
	}
	return fmt.Sprintf("%s%d.%02d", sign, c/100, c%100)
}

func buildLines() []Line {
	ls := []Line{
		{1001, 1, 1, 10, "18.00", "0.15", "153.00"},
		{1001, 2, 2, 3, "19.50", "9.99", "-525.92"},
		{1001, 3, 3, 1, "10.00", "0.00", "10.00"},
		{1002, 0, 4, 2, "5.00", "0.05", "9.50"},
		{1002, 5, 1, 1, "18.00", "0.00", "18.00"},
	}
	for _, o := range []struct{ ord, n int }{{1003, 46}, {1004, 40}} {
		tot := 0
		for i := 1; i <= o.n; i++ {
			ext := i * 125
			tot += ext
			ls = append(ls, Line{o.ord, i, (i-1)%3 + 1, i, "1.25", "0.00", cents(ext)})
		}
		for k := range Headers {
			if Headers[k].OrdNo == o.ord {
				Headers[k].Totl = cents(tot)
			}
		}
	}
	ls = append(ls,
		Line{9999999, 1, 99998, 99999, "9999999.99", "0.99", "999999999.99"},
		Line{9999999, 2, 1, 0, "0.00", "-0.25", "-123.45"},
		Line{9999999, 999, 99999, -5, "-1.00", "0.00", "200.00"},
	)
	return ls
}

// Q quotes an SQL character literal.
func Q(s string) string { return "'" + strings.ReplaceAll(s, "'", "''") + "'" }
